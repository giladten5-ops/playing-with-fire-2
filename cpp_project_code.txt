# ==============================================================================
# Playing With Fire 2 — Build & Run Makefile
# ==============================================================================

# Automatically prefer the project virtualenv Python if it exists
ifneq ($(wildcard ./venv/bin/python),)
    PYTHON ?= ./venv/bin/python
else
    PYTHON ?= python3
endif

CXX      ?= g++
CXXFLAGS ?= -O3 -Wall -std=c++17 -fPIC -Iinclude -MMD -MP

# Query pybind11 include flags and extension suffix from the active Python
PYBIND_INCLUDES := $(shell $(PYTHON) -m pybind11 --includes)
EXT_SUFFIX      := $(shell $(PYTHON) -c "import sysconfig; print(sysconfig.get_config_var('EXT_SUFFIX') or '.so')")

TARGET_SO  := bomberman_engine$(EXT_SUFFIX)
TARGET_BIN := bomberman
BUILD_DIR  := build

# Shared C++ engine sources (used by both Python extension and console binary)
ENGINE_SRCS := \
	src/core/Board.cpp \
	src/core/ConsoleInput.cpp \
	src/core/ConsoleRenderer.cpp \
	src/core/Game.cpp \
	src/core/World.cpp \
	src/entities/Entity.cpp \
	src/entities/Player.cpp \
	src/entities/Bomb.cpp \
	src/entities/Flame.cpp \
	src/entities/BreakableWall.cpp \
	src/entities/DumbBot.cpp \
	src/entities/PowerUp.cpp \
	src/entities/BombPowerUp.cpp \
	src/entities/LightningPowerUp.cpp \
	src/entities/SpeedPowerUp.cpp \
	src/entities/HammerPowerUp.cpp \
	src/entities/RingPowerUp.cpp \
	src/factories/PowerUpFactory.cpp \
	src/levels/ClassicLevel.cpp \
	src/systems/CollisionSystem.cpp \
	src/systems/ExplosionSystem.cpp \
	src/systems/SuddenDeathSystem.cpp

BINDING_SRC := src/bindings/bindings.cpp
MAIN_SRC    := src/main.cpp

ENGINE_OBJS  := $(patsubst src/%.cpp,$(BUILD_DIR)/%.o,$(ENGINE_SRCS))
BINDING_OBJ  := $(patsubst src/%.cpp,$(BUILD_DIR)/%.o,$(BINDING_SRC))
MAIN_OBJ     := $(patsubst src/%.cpp,$(BUILD_DIR)/%.o,$(MAIN_SRC))

ALL_DEPS := $(ENGINE_OBJS:.o=.d) $(BINDING_OBJ:.o=.d) $(MAIN_OBJ:.o=.d)

.PHONY: all build run play duel test train console clean rebuild help

# Default target: build the C++ Pybind11 engine module
all: build

build: $(TARGET_SO)

# Link the shared library for Python (bomberman_engine*.so)
$(TARGET_SO): $(ENGINE_OBJS) $(BINDING_OBJ)
	@echo "[Link] Creating Python C++ extension: $@"
	$(CXX) $(CXXFLAGS) -shared $^ -o $@
	@echo "[Done] Build complete: $@"

# Compile C++ engine object files
$(BUILD_DIR)/%.o: src/%.cpp
	@mkdir -p $(dir $@)
	@echo "[CXX]  $<"
	$(CXX) $(CXXFLAGS) $(PYBIND_INCLUDES) -c $< -o $@

# Build (if needed) and launch the Pygame GUI (starts in Main Menu)
run: $(TARGET_SO)
	$(PYTHON) play_gui.py

play: run

# Launch Pygame GUI directly in 1v1 AI Duel mode: v8 (Masked PPO) vs v7 (Grandmaster)
duel: $(TARGET_SO)
	$(PYTHON) play_gui.py --duel

# Build (if needed) and run the test suite
test: $(TARGET_SO)
	$(PYTHON) -m unittest discover -s tests -v

# Build (if needed) and run PPO RL training
train: $(TARGET_SO)
	$(PYTHON) train.py

# Build standalone terminal/console C++ executable
console: $(TARGET_BIN)

$(TARGET_BIN): $(ENGINE_OBJS) $(MAIN_OBJ)
	@echo "[Link] Creating console executable: $@"
	$(CXX) $(CXXFLAGS) $^ -o $@
	@echo "[Done] Run with ./$(TARGET_BIN)"

# Remove all build artifacts
clean:
	@echo "[Clean] Removing build artifacts..."
	rm -rf $(BUILD_DIR) *.o $(TARGET_BIN) bomberman_engine*.so
	find . -maxdepth 2 -name "__pycache__" -type d -exec rm -rf {} +

# Full clean + parallel rebuild
rebuild: clean
	$(MAKE) all

help:
	@echo "=================================================================="
	@echo " Playing With Fire 2 — Makefile Commands"
	@echo "=================================================================="
	@echo "  make           Compile the C++ engine module (incremental)"
	@echo "  make -j        Compile in parallel using all CPU cores"
	@echo "  make run       Compile (if needed) and launch the Pygame GUI"
	@echo "  make test      Compile (if needed) and run unit tests"
	@echo "  make train     Compile (if needed) and start RL training"
	@echo "  make console   Compile the standalone C++ terminal binary"
	@echo "  make clean     Remove all compiled object files and .so binaries"
	@echo "  make rebuild   Clean and recompile everything from scratch"
	@echo "=================================================================="

-include $(ALL_DEPS)

#include "../interfaces/IInput.h"

enum class EntityType {
    PLAYER,
    BOMB,
    FLAME,
    BREAKABLE_WALL,
    POWER_UP
};

class World;

class Entity {
private:
    int m_x;
    int m_y;
    bool m_isAlive;

protected:
    virtual void setX(int x);
    virtual void setY(int y);

public:
    Entity(int startX, int startY);
    virtual ~Entity() = default;

    virtual int getX() const;
    virtual int getY() const;
    virtual bool isAlive() const;
    virtual void destroy();

    virtual char getDisplayChar() const = 0;
    virtual EntityType getType() const = 0;
    virtual void update(World& world) = 0;

    virtual bool isSolid() const { return false; }
    virtual bool stopsExplosion() const { return false; }
    virtual bool shouldExplode() const { return false; }

    virtual void handleCommand(Command /*cmd*/, World& /*world*/) {}
    virtual void onExplosionHit(World& world);
    virtual void onIntersect(Entity& /*other*/, World& /*world*/) {}
};

CellType Board::getCellType(int x, int y) const {
    if (x < 0 || x >= m_width || y < 0 || y >= m_height) {
        return CellType::SOLID_WALL; // Out of bounds, treat as solid wall

        return; // Out of bounds, do nothing
    }
    m_grid[x][y] = type;
}

ConsoleInput::ConsoleInput() {
    setNonBlocking(true);
}

ConsoleInput::~ConsoleInput() {
    setNonBlocking(false);
}

void ConsoleInput::setNonBlocking(bool enable) {
    if (enable && !m_isNonBlocking) {
        tcgetattr(STDIN_FILENO, &m_oldTermios);
        struct termios newt = m_oldTermios;
        newt.c_lflag &= ~(ICANON | ECHO);
        tcsetattr(STDIN_FILENO, TCSANOW, &newt);
        int flags = fcntl(STDIN_FILENO, F_GETFL, 0);
        fcntl(STDIN_FILENO, F_SETFL, flags | O_NONBLOCK);
        m_isNonBlocking = true;
    } else if (!enable && m_isNonBlocking) {
        tcsetattr(STDIN_FILENO, TCSANOW, &m_oldTermios);
        int flags = fcntl(STDIN_FILENO, F_GETFL, 0);
        fcntl(STDIN_FILENO, F_SETFL, flags & ~O_NONBLOCK);
        m_isNonBlocking = false;
    }
}

char ConsoleInput::getRawChar() {
    char c = '\0';
    if (read(STDIN_FILENO, &c, 1) == 1) {
        return c;
    }
    return '\0';
}

Command ConsoleInput::getCommand() {
    char c = getRawChar();
    switch (c) {
        case 'w': case 'W': return Command::UP;
        case 's': case 'S': return Command::DOWN;
        case 'a': case 'A': return Command::LEFT;
        case 'd': case 'D': return Command::RIGHT;
        case ' ': return Command::PLACE_BOMB;
        case 'q': case 'Q': return Command::QUIT;
        default: return Command::NONE;
    }
}

std::vector<PlayerCommand> ConsoleInput::pollCommands() {
    std::vector<PlayerCommand> commands;
    char pending = '\0';

    while (true) {
        char input = (pending != '\0') ? pending : getRawChar();
        pending = '\0';
        if (input == '\0') break;

        if (input == 'q' || input == 'Q') {
            commands.push_back({0, Command::QUIT});
            break;
        }

        if (input == '\033') {
            char next = getRawChar();
            if (next == '[') {
                char arrow = getRawChar();
                if (arrow == 'A') commands.push_back({1, Command::UP});
                else if (arrow == 'B') commands.push_back({1, Command::DOWN});
                else if (arrow == 'C') commands.push_back({1, Command::RIGHT});
                else if (arrow == 'D') commands.push_back({1, Command::LEFT});
            } else if (next != '\0') {
                pending = next;
            }
            continue;
        }

        switch (input) {
            case ' ': commands.push_back({1, Command::PLACE_BOMB}); break;
            case 'w': case 'W': commands.push_back({2, Command::UP}); break;
            case 's': case 'S': commands.push_back({2, Command::DOWN}); break;
            case 'a': case 'A': commands.push_back({2, Command::LEFT}); break;
            case 'd': case 'D': commands.push_back({2, Command::RIGHT}); break;
            case 'e': case 'E': commands.push_back({2, Command::PLACE_BOMB}); break;
            default: break;
        }
    }
    return commands;
}

namespace {
    int getRenderPriority(EntityType type) {
        switch (type) {
            case EntityType::POWER_UP:       return 1;
            case EntityType::BOMB:           return 2;
            case EntityType::BREAKABLE_WALL: return 3;
            case EntityType::PLAYER:         return 4;
            case EntityType::FLAME:          return 5;
        }
        return 0;
    }
}

void ConsoleRenderer::render(const World& world, int sdTimerTicks, bool sdActive) {
    const Board& board = world.getBoard();
    int width = board.getWidth();
    int height = board.getHeight();

    std::vector<std::vector<char>> display(width, std::vector<char>(height, ' '));
    std::vector<std::vector<int>> priority(width, std::vector<int>(height, 0));

    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            CellType cellType = board.getCellType(x, y);
            if (cellType == CellType::SOLID_WALL) {
                display[x][y] = '#';
                priority[x][y] = 10;
            } else if (cellType == CellType::BREAKABLE_WALL) {
                display[x][y] = 'B';
                priority[x][y] = 3;
            }
        }
    }

    for (const auto& entity : world.getEntities()) {
        if (!entity->isAlive()) continue;

        int ex = entity->getX();
        int ey = entity->getY();

        if (ex >= 0 && ex < width && ey >= 0 && ey < height) {
            char ch = entity->getDisplayChar();
            if (ch == ' ') continue;

            int p = getRenderPriority(entity->getType());
            if (p >= priority[ex][ey]) {
                display[ex][ey] = ch;
                priority[ex][ey] = p;
            }
        }
    }

    std::cout << "\033[2J\033[1;1H";

    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            std::cout << display[x][y] << ' ' << ' ';
        }
        std::cout << std::endl;
    }

    std::cout << "\n============================================\n";
    
    // Display Sudden Death countdown or active alert
    if (sdActive) {
        std::cout << "      !!! SUDDEN DEATH ACTIVE !!!\n";
    } else if (sdTimerTicks >= 0) {
        int seconds = sdTimerTicks / 30;
        std::cout << "      Sudden Death in: " << seconds << "s\n";
    }
    std::cout << "--------------------------------------------\n";

    for (const auto& entity : world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::PLAYER) {
            const Player* p = static_cast<const Player*>(entity.get());

            std::cout << " " << p->getDisplayName() << " | ";
            std::cout << "Lives: " << p->getLives() << " | ";
            std::cout << "Bombs: " << p->getMaxBombs() << " | ";
            std::cout << "Rad: " << p->getBombRadius() << " | ";
            std::cout << "Spd: " << p->getSpeedLevel();

            if (p->hasHammer()) std::cout << " [H]";
            if (p->hasRing()) std::cout << " [R]";

            std::cout << "\n";
        }
    }
    std::cout << "============================================\n";
}

void Game::loadLevel(ILevel& level, int numHumanPlayers, int numBots, bool shuffleSpawns) {
    level.generate(m_world);
    if (numHumanPlayers > 0 || numBots > 0) {
        auto spawnPoints = level.getSpawnPoints();
        if (shuffleSpawns && spawnPoints.size() > 1) {
            for (size_t i = spawnPoints.size() - 1; i > 0; --i) {
                size_t j = static_cast<size_t>(std::rand()) % (i + 1);
                std::swap(spawnPoints[i], spawnPoints[j]);
            }
        }
        size_t currentSpawnIdx = 0;
        int nextId = 1;
        for (int i = 0; i < numHumanPlayers && currentSpawnIdx < spawnPoints.size(); ++i) {
            m_world.addEntity(std::make_unique<Player>(
                spawnPoints[currentSpawnIdx].x, spawnPoints[currentSpawnIdx].y, nextId++));
            currentSpawnIdx++;
        }
        for (int i = 0; i < numBots && currentSpawnIdx < spawnPoints.size(); ++i) {
            m_world.addEntity(std::make_unique<DumbBot>(
                spawnPoints[currentSpawnIdx].x, spawnPoints[currentSpawnIdx].y, nextId++));
            currentSpawnIdx++;
        }
        m_world.sync();
    }
}

void Game::update(Command command) {
    std::vector<PlayerCommand> commands;
    if (command != Command::NONE) commands.push_back({1, command});
    update(commands);
}

void Game::update(const std::vector<PlayerCommand>& commands) {
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive()) entity->update(m_world);
    }
    m_world.sync();

    if (!commands.empty()) {
        for (const auto& pc : commands) {
            Player* p = m_world.getPlayer(pc.playerId);
            if (p && p->isAlive() && !p->isBot()) {
                p->handleCommand(pc.command, m_world);
            }
        }
        m_world.sync();
    }

    m_explosionSystem.process(m_world);
    m_collisionSystem.process(m_world);
    m_suddenDeathSystem.process(m_world);
    m_world.sync();
}

bool Game::isGameOver() const {
    int alivePlayers = 0;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::PLAYER) alivePlayers++;
    }
    return alivePlayers <= 1;
}

int Game::getWinnerId() const {
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::PLAYER) {
            return static_cast<const Player*>(entity.get())->getId();
        }
    }
    return -1;
}

    int height = m_world.getBoard().getHeight();
    m_suddenDeathSystem = SuddenDeathSystem(80 * 30, width, height);

PlayerInfo Game::getPlayerInfo(int playerId) const {
    PlayerInfo info;
    info.id = playerId;
    const Player* p = m_world.getPlayer(playerId);
    if (!p || !p->isAlive()) {
        return info;
    }
    info.x = p->getX();
    info.y = p->getY();
    info.lives = p->getLives();
    info.maxBombs = p->getMaxBombs();
    int activeB = m_world.countBombsForOwner(playerId);
    info.availBombs = std::max(0, info.maxBombs - activeB);
    info.bombRadius = p->getBombRadius();
    info.speedLevel = p->getSpeedLevel();
    info.cooldown = p->getCooldown();
    info.moveCooldownMax = p->getMoveCooldownMax();
    info.invincibilityTimer = p->getInvincibilityTimer();
    info.facingDx = p->getFacingDx();
    info.facingDy = p->getFacingDy();
    info.hasHammer = p->hasHammer();
    info.hasRing = p->hasRing();
    info.isBot = p->isBot();
    info.isAlive = p->isAlive();
    return info;
}

std::vector<BombInfo> Game::getBombs() const {
    std::vector<BombInfo> bombs;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
            const auto* b = static_cast<const Bomb*>(entity.get());
            BombInfo bi;
            bi.x = b->getX();
            bi.y = b->getY();
            bi.timer = b->getTimer();
            bi.radius = b->getRadius();
            bi.ownerId = b->getOwnerId();
            bi.creditOwnerId = b->getCreditOwnerId();
            bi.isSliding = b->isSliding();
            bi.slideDx = b->getSlideDx();
            bi.slideDy = b->getSlideDy();
            bi.slideCooldown = b->getSlideCooldown();
            bi.slideCooldownMax = Bomb::SLIDE_COOLDOWN_FRAMES;
            bombs.push_back(bi);
        }
    }
    return bombs;
}

void Game::givePlayerPowerUp(int playerId, const std::string& type) {
    Player* p = m_world.getPlayer(playerId);

    if (type == "hammer" || type == "H") p->enableHammer();
    else if (type == "ring" || type == "R") p->enableRing();

    else if (type == "lightning" || type == "L") p->increaseBombRadius();
    else if (type == "speed" || type == "S") p->increaseSpeed();
}

void Game::fillState(int playerId, float* state) const {
    const Board& board = m_world.getBoard();
    int width = board.getWidth();
    int height = board.getHeight();

    auto setPixel = [&](int c, int x, int y, float value) {
        if (x >= 0 && x < width && y >= 0 && y < height) {
            int index = c * width * height + y * width + x;
            state[index] = std::max(state[index], value);
        }
    };
    auto getPixel = [&](int c, int x, int y) -> float {
        if (x >= 0 && x < width && y >= 0 && y < height) {
            return state[c * width * height + y * width + x];
        }
        return 0.0f;
    };

    for (int x = 0; x < width; x++) {
        for (int y = 0; y < height; y++) {
            CellType ct = board.getCellType(x, y);
            if (ct == CellType::SOLID_WALL) setPixel(0, x, y, 1.0f);
            else if (ct == CellType::BREAKABLE_WALL) setPixel(1, x, y, 1.0f);
        }
    }

    const Player* selfPlayer = m_world.getPlayer(playerId);
    if (selfPlayer) {
        int maxB = selfPlayer->getMaxBombs();
        int activeB = m_world.countBombsForOwner(playerId);
        float availB = std::max(0, maxB - activeB) / 10.0f;
        float specialVal = (selfPlayer->hasHammer() ? 0.6f : 0.0f) + (selfPlayer->hasRing() ? 0.4f : 0.0f);
        for (int i = 0; i < width * height; ++i) {
            state[8 * width * height + i] = availB;
            state[9 * width * height + i] = specialVal;
        }
    }

    std::vector<const Bomb*> activeBombs;
    for (const auto& entity : m_world.getEntities()) {
        if (!entity->isAlive()) continue;
        int ex = entity->getX(), ey = entity->getY();
        EntityType type = entity->getType();

        if (type == EntityType::FLAME) { setPixel(4, ex, ey, 1.0f); setPixel(3, ex, ey, 1.0f); }
        else if (type == EntityType::POWER_UP) setPixel(5, ex, ey, 1.0f);
        else if (type == EntityType::PLAYER) {
            const Player* player = static_cast<const Player*>(entity.get());
            if (player->getId() == playerId) {
                setPixel(6, ex, ey, player->isOnCooldown() ? 0.5f : 1.0f);
            } else {
                setPixel(7, ex, ey, 1.0f);
            }
        } else if (type == EntityType::BOMB) {
            activeBombs.push_back(static_cast<const Bomb*>(entity.get()));
        }
    }

    const int dx[] = {1, -1, 0, 0};
    const int dy[] = {0, 0, 1, -1};

    // Chain-Reaction Timer Propagation across bombs in line-of-fire
    int nBombs = static_cast<int>(activeBombs.size());
    std::vector<int> effectiveTimers(nBombs);
    for (int i = 0; i < nBombs; ++i) {
        effectiveTimers[i] = activeBombs[i]->getTimer();
    }

    bool changed = true;
    int maxIter = nBombs + 1;
    while (changed && maxIter-- > 0) {
        changed = false;
        for (int i = 0; i < nBombs; ++i) {
            int bx = activeBombs[i]->getX();
            int by = activeBombs[i]->getY();
            int radius = activeBombs[i]->getRadius();
            int t_i = effectiveTimers[i];

            for (int dir = 0; dir < 4; ++dir) {
                for (int r = 1; r <= radius; ++r) {
                    int nx = bx + dx[dir] * r;
                    int ny = by + dy[dir] * r;
                    if (!m_world.isPositionValid(nx, ny) || board.getCellType(nx, ny) == CellType::SOLID_WALL) break;
                    if (getPixel(1, nx, ny) > 0.0f) break; // Breakable wall blocks chain

                    bool hitOtherBomb = false;
                    for (int j = 0; j < nBombs; ++j) {
                        if (i != j && activeBombs[j]->getX() == nx && activeBombs[j]->getY() == ny) {
                            if (t_i < effectiveTimers[j]) {
                                effectiveTimers[j] = t_i;
                                changed = true;
                            }
                            hitOtherBomb = true;
                        }
                    }
                    if (hitOtherBomb) break;
                }
            }
        }
    }

    for (int bIdx = 0; bIdx < nBombs; ++bIdx) {
        const Bomb* bomb = activeBombs[bIdx];
        int ex = bomb->getX(), ey = bomb->getY();
        float tNorm = std::clamp(1.0f - (static_cast<float>(effectiveTimers[bIdx]) / 100.0f), 0.1f, 1.0f);
        setPixel(2, ex, ey, tNorm);
        setPixel(3, ex, ey, tNorm);

        auto projectBlastFrom = [&](int cx, int cy, float urgency) {
            setPixel(3, cx, cy, urgency);
            for (int i = 0; i < 4; ++i) {
                for (int r = 1; r <= bomb->getRadius(); ++r) {
                    int nx = cx + dx[i] * r, ny = cy + dy[i] * r;
                    if (!m_world.isPositionValid(nx, ny) || board.getCellType(nx, ny) == CellType::SOLID_WALL) break;
                    setPixel(3, nx, ny, urgency);
                    if (getPixel(1, nx, ny) > 0.0f) break;
                }
            }
        };

        projectBlastFrom(ex, ey, tNorm);

        // If the bomb is sliding, project its slide trajectory and final stopping blast zone
        if (bomb->isSliding()) {
            int sdx = bomb->getSlideDx();
            int sdy = bomb->getSlideDy();
            int curX = ex;
            int curY = ey;
            int ticksLeft = std::max(0, effectiveTimers[bIdx]);
            int stepTicks = Bomb::SLIDE_COOLDOWN_FRAMES + 1;
            int maxSlideSteps = std::max(1, (ticksLeft + stepTicks - 1) / stepTicks);

            for (int s = 1; s <= maxSlideSteps; ++s) {
                int nextX = curX + sdx;
                int nextY = curY + sdy;
                if (!Bomb::canSlideTo(m_world, nextX, nextY)) {
                    break;
                }
                curX = nextX;
                curY = nextY;
                float slideDanger = std::clamp(std::max(tNorm, 0.72f - 0.04f * s), 0.25f, 1.0f);
                setPixel(3, curX, curY, slideDanger);
            }
            if (curX != ex || curY != ey) {
                projectBlastFrom(curX, curY, tNorm);
            }
        }
    }

    // Imminent Sudden Death spiral danger projection into Channel 3 (danger)
    auto sdTiles = m_suddenDeathSystem.getImminentDangerTiles(90);
    for (const auto& sdt : sdTiles) {
        float sdUrgency = std::clamp(1.0f - (static_cast<float>(sdt.ticksUntilDrop) / 95.0f), 0.25f, 1.0f);
        setPixel(3, sdt.x, sdt.y, sdUrgency);
    }
}

std::vector<float> Game::getState(int playerId) const {
    const Board& board = m_world.getBoard();
    int width = board.getWidth();
    int height = board.getHeight();
    int channels = 10;

    std::vector<float> state(channels * width * height, 0.0f);
    fillState(playerId, state.data());
    return state;
}

StepResult Game::step(int playerId, Command action, const std::vector<PlayerCommand>& extraCommands) {
    StepResult result;
    result.isDone = false;
    result.reward = -0.01f;

    const Player* player = m_world.getPlayer(playerId);
    bool wasAlive = player && player->isAlive();
    int oldX = wasAlive ? player->getX() : -1;
    int oldY = wasAlive ? player->getY() : -1;
    int livesBefore = wasAlive ? player->getLives() : 0;
    int maxBombsBefore = wasAlive ? player->getMaxBombs() : 0;
    int radiusBefore = wasAlive ? player->getBombRadius() : 0;
    int speedBefore = wasAlive ? player->getSpeedLevel() : 0;
    bool hammerBefore = wasAlive ? player->hasHammer() : false;
    bool ringBefore = wasAlive ? player->hasRing() : false;
    int bombsOwnedBefore = m_world.countBombsForOwner(playerId);

    int slidingBefore = 0;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
            const auto* b = static_cast<const Bomb*>(entity.get());
            if (b->isSliding() && b->getCreditOwnerId() == playerId) slidingBefore++;
        }
    }

    m_world.popWallsDestroyed(playerId);
    m_world.popEnemiesHit(playerId);
    m_world.popEnemiesKilled(playerId);

    std::vector<PlayerCommand> commands;
    if (action != Command::NONE) commands.push_back({playerId, action});
    for (const auto& ec : extraCommands) {
        if (ec.playerId != playerId && ec.command != Command::NONE) {
            commands.push_back(ec);
        }
    }
    update(commands);

    if (action == Command::PLACE_BOMB && m_world.countBombsForOwner(playerId) > bombsOwnedBefore) {
        result.bombPlaced = true;
    }

    int slidingAfter = 0;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
            const auto* b = static_cast<const Bomb*>(entity.get());
            if (b->isSliding() && b->getCreditOwnerId() == playerId) slidingAfter++;
        }
    }
    if (slidingAfter > slidingBefore) {
        result.bombPushed = true;
    }

    int wallsDestroyedByPlayer = m_world.popWallsDestroyed(playerId);
    result.wallsDestroyed = wallsDestroyedByPlayer;
    if (wallsDestroyedByPlayer > 0) result.reward += wallsDestroyedByPlayer * 3.0f;

    player = m_world.getPlayer(playerId);
    bool isAliveNow = player && player->isAlive();

    if (isAliveNow) {
        result.moved = (player->getX() != oldX || player->getY() != oldY);
        if (player->getLives() < livesBefore) {
            result.livesLost = (livesBefore - player->getLives());
            result.reward -= result.livesLost * 25.0f;
        }

        int powerUpsGained = 0;
        if (player->getMaxBombs() > maxBombsBefore) powerUpsGained++;
        if (player->getBombRadius() > radiusBefore) powerUpsGained++;
        if (player->getSpeedLevel() > speedBefore) powerUpsGained++;
        if (player->hasHammer() && !hammerBefore) powerUpsGained++;
        if (player->hasRing() && !ringBefore) powerUpsGained++;

        result.powerUpsCollected = powerUpsGained;
        if (powerUpsGained > 0) result.reward += powerUpsGained * 8.0f;
    } else if (wasAlive) {
        result.livesLost = livesBefore;
        result.reward -= result.livesLost * 25.0f;
    }

    int enemiesHitByPlayer = m_world.popEnemiesHit(playerId);
    if (enemiesHitByPlayer > 0) {
        result.enemiesHit = enemiesHitByPlayer;
        result.reward += result.enemiesHit * 10.0f;
    }

    int enemiesKilledByPlayer = m_world.popEnemiesKilled(playerId);
    if (enemiesKilledByPlayer > 0) {
        result.enemiesKilled = enemiesKilledByPlayer;
        result.reward += result.enemiesKilled * 20.0f;
    }

    result.isDone = isGameOver();
    if (wasAlive && !isAliveNow) {
        result.died = true;
        result.reward -= 100.0f;
        result.isDone = true;
    } else if (result.isDone && isAliveNow) {
        result.won = true;
        result.reward += 100.0f;
    }

    return result;
}

std::vector<PowerUpInfo> Game::getPowerUps() const {
    std::vector<PowerUpInfo> powerUps;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::POWER_UP) {
            PowerUpInfo pui;
            pui.x = entity->getX();
            pui.y = entity->getY();
            pui.type = entity->getDisplayChar(); 
            powerUps.push_back(pui);
        }
    }
    return powerUps;
}

std::vector<FlameInfo> Game::getFlames() const {
    std::vector<FlameInfo> flames;
    for (const auto& entity : m_world.getEntities()) {
        if (entity->isAlive() && entity->getType() == EntityType::FLAME) {
            const auto* flame = static_cast<const Flame*>(entity.get());
            FlameInfo fi;
            fi.x = flame->getX();
            fi.y = flame->getY();
            fi.timer = flame->getTimer();
            fi.ownerId = flame->getOwnerId();
            flames.push_back(fi);
        }
    }
    return flames;
}

std::vector<std::vector<int>> Game::getBoardGrid() const {
    const Board& board = m_world.getBoard();
    int width = board.getWidth();
    int height = board.getHeight();
    std::vector<std::vector<int>> grid(height, std::vector<int>(width, 0));

    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            CellType ct = board.getCellType(x, y);
            if (ct == CellType::SOLID_WALL) {
                grid[y][x] = 1;
            } else if (ct == CellType::BREAKABLE_WALL) {
                grid[y][x] = 2;
            }
        }
    }
    return grid;
}

void World::addEntity(std::unique_ptr<Entity> entity) {
    if (entity) {
        m_spawnQueue.push_back(std::move(entity));
    }
}

void World::sync() {
    for (auto& entity : m_spawnQueue) {
        m_entities.push_back(std::move(entity));
    }
    m_spawnQueue.clear();

    m_entities.erase(
        std::remove_if(m_entities.begin(), m_entities.end(),
            [](const std::unique_ptr<Entity>& entity) {
                return !entity->isAlive();
            }),
        m_entities.end()
    );
}

bool World::isPositionValid(int x, int y) const {
    return x >= 0 && x < m_board.getWidth() && y >= 0 && y < m_board.getHeight();
}

bool World::isTileFree(int x, int y) const {
    if (!isPositionValid(x, y)) {
        return false;
    }

    if (m_board.getCellType(x, y) != CellType::EMPTY) {
        return false;
    }

    for (const auto& entity : m_entities) {
        if (entity->isAlive() && entity->getX() == x && entity->getY() == y && entity->isSolid()) {
            return false;
        }
    }
    for (const auto& entity : m_spawnQueue) {
        if (entity->isAlive() && entity->getX() == x && entity->getY() == y && entity->isSolid()) {
            return false;
        }
    }
    return true;
}

int World::countBombsForOwner(int ownerId) const {
    int count = 0;
    for (const auto& entity : m_entities) {
        if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
            if (static_cast<const Bomb*>(entity.get())->getOwnerId() == ownerId) count++;
        }
    }
    for (const auto& entity : m_spawnQueue) {
        if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
            if (static_cast<const Bomb*>(entity.get())->getOwnerId() == ownerId) count++;
        }
    }
    return count;
}

            if (p->getId() == id) return p;
        }
    }
    return nullptr;
}

            const auto* p = static_cast<const Player*>(e.get());
            if (p->getId() == id) return p;

}

    m_entities.clear();
    m_spawnQueue.clear();
    m_wallsDestroyedByOwner.clear();
    m_enemiesKilledByOwner.clear();
    m_enemiesHitByOwner.clear();

    int width = m_board.getWidth();
    int height = m_board.getHeight();

    for (int x = 0; x < width; x++) {
        for (int y = 0; y < height; y++) {
            m_board.setCellType(x, y, CellType::EMPTY);
        }
    }
}

                count++;
            }
        }
    }
    return count;
}

bool Bomb::canSlideTo(const World& world, int x, int y) {
    if (!world.isTileFree(x, y)) {
        return false;
    }
    for (const auto* e : world.getEntitiesAt(x, y)) {
        if (e->isAlive() && (e->getType() == EntityType::PLAYER || e->getType() == EntityType::FLAME)) {
            return false;
        }
    }
    return true;
}

void Bomb::update(World& world) {
    if (m_dx != 0 || m_dy != 0) {
        if (m_slideCooldown > 0) {
            m_slideCooldown--;
        } else {
            int nextX = getX() + m_dx;
            int nextY = getY() + m_dy;
            if (canSlideTo(world, nextX, nextY)) {
                setX(nextX);
                setY(nextY);
                m_slideCooldown = SLIDE_COOLDOWN_FRAMES;
            } else {
                m_dx = 0;
                m_dy = 0;
                m_slideCooldown = 0;
            }
        }
    }

    if (!isAlive()) return;
    if (m_timer > 0) {
        m_timer--;
    }
}

    return isAlive() && m_timer <= 0;

    if (isAlive()) {
        m_timer = 0;
    }
}

    m_dy = dy;

bool DumbBot::canStepOn(int x, int y, const World& world) const {
    if (!world.isPositionValid(x, y)) return false;
    if (world.getBoard().getCellType(x, y) != CellType::EMPTY) return false;

    for (const auto* e : world.getEntitiesAt(x, y)) {
        if (!e->isAlive()) continue;
        if (e->getType() == EntityType::FLAME) {
            return false;
        }
        if (e->isSolid()) {
            if (e->getType() == EntityType::BOMB) {
                const auto* bomb = static_cast<const Bomb*>(e);
                if (!hasRing() || hasHammer() || bomb->getTimer() <= 20) {
                    return false;
                }
            } else {
                return false;
            }
        }
    }
    return true;
}

bool DumbBot::hasClearBlastPath(int bx, int by, int tx, int ty, const World& world) const {
    int dx = (tx > bx) ? 1 : ((tx < bx) ? -1 : 0);
    int dy = (ty > by) ? 1 : ((ty < by) ? -1 : 0);

    int cx = bx + dx;
    int cy = by + dy;
    while (cx != tx || cy != ty) {
        if (world.getBoard().getCellType(cx, cy) != CellType::EMPTY) return false;
        auto entities = world.getEntitiesAt(cx, cy);
        for (const auto* e : entities) {
            if (e->isAlive() && e->stopsExplosion() && e->getType() != EntityType::BOMB) {
                return false;
            }
        }
        cx += dx;
        cy += dy;
    }
    return true;
}

int DumbBot::getTileTimeToExplode(int x, int y, const World& world, int extraBx, int extraBy, int extraRadius) const {
    if (!world.isPositionValid(x, y)) return 0;

    struct BombData {
        int x;
        int y;
        int radius;
        int timer;
    };
    std::vector<BombData> bombs;

    auto collectFromList = [&](const std::vector<std::unique_ptr<Entity>>& list) {
        for (const auto& entity : list) {
            if (!entity->isAlive()) continue;
            if (entity->getType() == EntityType::FLAME) {
                if (entity->getX() == x && entity->getY() == y) {
                    return true;
                }
            } else if (entity->getType() == EntityType::BOMB) {
                const auto* b = static_cast<const Bomb*>(entity.get());
                bombs.push_back({b->getX(), b->getY(), b->getRadius(), b->getTimer()});
                if (b->isSliding()) {
                    int sdx = b->getSlideDx();
                    int sdy = b->getSlideDy();
                    int sx = b->getX();
                    int sy = b->getY();
                    for (int s = 1; s <= 4; ++s) {
                        int nx = sx + sdx;
                        int ny = sy + sdy;
                        if (!Bomb::canSlideTo(world, nx, ny)) break;
                        sx = nx;
                        sy = ny;
                        bombs.push_back({sx, sy, b->getRadius(), std::max(1, b->getTimer())});
                    }
                }
            }
        }
        return false;
    };

    if (collectFromList(world.getEntities()) || collectFromList(world.getSpawnQueue())) {
        return 0;
    }

    if (extraBx >= 0 && extraBy >= 0) {
        bombs.push_back({extraBx, extraBy, extraRadius, DEFAULT_BOMB_TIMER});
    }

    if (bombs.empty()) return 9999;

    // Propagate chain-reaction timers across bombs in line-of-fire
    bool changed = true;
    int maxIter = static_cast<int>(bombs.size()) + 1;
    while (changed && maxIter-- > 0) {
        changed = false;
        for (size_t i = 0; i < bombs.size(); ++i) {
            for (size_t j = 0; j < bombs.size(); ++j) {
                if (i == j) continue;
                bool alignedX = (bombs[i].x == bombs[j].x && std::abs(bombs[i].y - bombs[j].y) <= bombs[i].radius);
                bool alignedY = (bombs[i].y == bombs[j].y && std::abs(bombs[i].x - bombs[j].x) <= bombs[i].radius);
                if ((alignedX || alignedY) && bombs[i].timer < bombs[j].timer) {
                    if (hasClearBlastPath(bombs[i].x, bombs[i].y, bombs[j].x, bombs[j].y, world)) {
                        bombs[j].timer = bombs[i].timer;
                        changed = true;
                    }
                }
            }
        }
    }

    int minTimer = 9999;
    for (const auto& b : bombs) {
        if (b.x == x && std::abs(b.y - y) <= b.radius) {
            if (hasClearBlastPath(b.x, b.y, x, y, world)) {
                minTimer = std::min(minTimer, b.timer);
            }
        }
        if (b.y == y && std::abs(b.x - x) <= b.radius) {
            if (hasClearBlastPath(b.x, b.y, x, y, world)) {
                minTimer = std::min(minTimer, b.timer);
            }
        }
    }
    return minTimer;
}

bool DumbBot::isTileDangerous(int x, int y, const World& world) const {
    return getTileTimeToExplode(x, y, world) < 9999;
}

bool DumbBot::isTileDangerousWithExtraBomb(int x, int y, const World& world, int extraBx, int extraBy, int extraRadius) const {
    return getTileTimeToExplode(x, y, world, extraBx, extraBy, extraRadius) < 9999;
}

Command DumbBot::findEscapeDirection(const World& world, int extraBx, int extraBy, int extraRadius) const {
    int width = world.getBoard().getWidth();
    int height = world.getBoard().getHeight();
    int startX = getX();
    int startY = getY();
    int moveCd = getMoveCooldownMax();

    struct StepNode {
        int x;
        int y;
        Command firstCommand;
        int dist;
    };

    std::vector<std::vector<bool>> visited(width, std::vector<bool>(height, false));
    std::queue<StepNode> q;
    visited[startX][startY] = true;

    struct DirInfo { int dx; int dy; Command cmd; };
    const DirInfo baseDirs[4] = {
        {0, -1, Command::UP},
        {0, 1, Command::DOWN},
        {-1, 0, Command::LEFT},
        {1, 0, Command::RIGHT}
    };

    // Order directions so m_currentDir is evaluated first (committing to active escape path),
    // and the opposite direction is evaluated last (avoiding oscillation).
    std::vector<DirInfo> orderedDirs;
    orderedDirs.reserve(4);
    Command oppositeDir = getOppositeCommand(m_currentDir);
    for (const auto& d : baseDirs) {
        if (d.cmd == m_currentDir) orderedDirs.push_back(d);
    }
    for (const auto& d : baseDirs) {
        if (d.cmd != m_currentDir && d.cmd != oppositeDir) orderedDirs.push_back(d);
    }
    for (const auto& d : baseDirs) {
        if (d.cmd == oppositeDir && d.cmd != m_currentDir) orderedDirs.push_back(d);
    }

    const int maxEscapeSteps = (extraBx >= 0) ? 3 : 6;

    for (const auto& d : orderedDirs) {
        int nx = startX + d.dx;
        int ny = startY + d.dy;
        if (canStepOn(nx, ny, world)) {
            int tte = getTileTimeToExplode(nx, ny, world, extraBx, extraBy, extraRadius);
            if (tte == 9999) {
                return d.cmd;
            }
            if (tte > moveCd + 3) {
                visited[nx][ny] = true;
                q.push({nx, ny, d.cmd, 1});
            }
        }
    }

    while (!q.empty()) {
        StepNode curr = q.front();
        q.pop();

        if (curr.dist >= maxEscapeSteps) continue;

        for (const auto& d : orderedDirs) {
            int nx = curr.x + d.dx;
            int ny = curr.y + d.dy;
            if (world.isPositionValid(nx, ny) && !visited[nx][ny] && canStepOn(nx, ny, world)) {
                int nextDist = curr.dist + 1;
                int tte = getTileTimeToExplode(nx, ny, world, extraBx, extraBy, extraRadius);
                visited[nx][ny] = true;
                if (tte == 9999) {
                    return curr.firstCommand;
                }
                if (tte > nextDist * moveCd + 3) {
                    q.push({nx, ny, curr.firstCommand, nextDist});
                }
            }
        }
    }

    return Command::NONE;
}

Command DumbBot::findExplorationDirection(const World& world) const {
    int cx = getX();
    int cy = getY();

    struct DirInfo { int dx; int dy; Command cmd; };
    const DirInfo baseDirs[4] = {
        {0, -1, Command::UP},
        {0, 1, Command::DOWN},
        {-1, 0, Command::LEFT},
        {1, 0, Command::RIGHT}
    };

    Command oppositeDir = getOppositeCommand(m_currentDir);
    std::vector<Command> safeMoves;
    std::vector<Command> forwardSafeMoves;

    for (const auto& d : baseDirs) {
        int nx = cx + d.dx;
        int ny = cy + d.dy;
        if (canStepOn(nx, ny, world) && !isTileDangerous(nx, ny, world)) {
            // Immediate priority: step onto adjacent power-up
            for (const auto* e : world.getEntitiesAt(nx, ny)) {
                if (e->isAlive() && e->getType() == EntityType::POWER_UP) {
                    return d.cmd;
                }
            }
            safeMoves.push_back(d.cmd);
            if (d.cmd != oppositeDir) {
                forwardSafeMoves.push_back(d.cmd);
            }
        }
    }

    if (safeMoves.empty()) {
        return Command::NONE;
    }

    // If this bot already has an active bomb on the board and is standing on a safe tile,
    // wait calmly for the bomb to detonate (only stepping 1 tile further if directly adjacent to a blast line).
    if (world.countBombsForOwner(getId()) > 0) {
        auto countDangerousNeighbors = [&](int tx, int ty) {
            int count = 0;
            for (const auto& d : baseDirs) {
                int wx = tx + d.dx;
                int wy = ty + d.dy;
                if (world.isPositionValid(wx, wy) && isTileDangerous(wx, wy, world)) {
                    count++;
                }
            }
            return count;
        };

        int curDangerAdj = countDangerousNeighbors(cx, cy);
        if (curDangerAdj > 0) {
            for (const auto& d : baseDirs) {
                int nx = cx + d.dx;
                int ny = cy + d.dy;
                if (canStepOn(nx, ny, world) && !isTileDangerous(nx, ny, world)) {
                    if (countDangerousNeighbors(nx, ny) == 0) {
                        return d.cmd;
                    }
                }
            }
        }
        return Command::NONE;
    }

    // Run a safe-BFS (depth <= 10) to locate the nearest reachable power-up, breakable wall, or open space
    int width = world.getBoard().getWidth();
    int height = world.getBoard().getHeight();
    std::vector<std::vector<bool>> visited(width, std::vector<bool>(height, false));
    struct Node { int x; int y; Command firstCmd; int dist; };
    std::queue<Node> q;
    visited[cx][cy] = true;
    int reachableSafeCount = 1;

    std::vector<DirInfo> orderedDirs;
    for (const auto& d : baseDirs) {
        if (d.cmd == m_currentDir) orderedDirs.push_back(d);
    }
    for (const auto& d : baseDirs) {
        if (d.cmd != m_currentDir && d.cmd != oppositeDir) orderedDirs.push_back(d);
    }
    for (const auto& d : baseDirs) {
        if (d.cmd == oppositeDir && d.cmd != m_currentDir) orderedDirs.push_back(d);
    }

    for (const auto& d : orderedDirs) {
        int nx = cx + d.dx;
        int ny = cy + d.dy;
        if (canStepOn(nx, ny, world) && !isTileDangerous(nx, ny, world)) {
            visited[nx][ny] = true;
            reachableSafeCount++;
            q.push({nx, ny, d.cmd, 1});
        }
    }

    Command wallTargetCmd = Command::NONE;
    Command enemyTargetCmd = Command::NONE;
    while (!q.empty()) {
        Node curr = q.front();
        q.pop();

        for (const auto* e : world.getEntitiesAt(curr.x, curr.y)) {
            if (e->isAlive() && e->getType() == EntityType::POWER_UP) {
                return curr.firstCmd;
            }
        }

        if (enemyTargetCmd == Command::NONE) {
            for (const auto& entity : world.getEntities()) {
                if (entity->isAlive() && entity->getType() == EntityType::PLAYER && entity.get() != this) {
                    int dist = std::abs(entity->getX() - curr.x) + std::abs(entity->getY() - curr.y);
                    if (dist <= 2) {
                        enemyTargetCmd = curr.firstCmd;
                        break;
                    }
                }
            }
        }

        if (wallTargetCmd == Command::NONE) {
            for (const auto& d : baseDirs) {
                int wx = curr.x + d.dx;
                int wy = curr.y + d.dy;
                if (world.isPositionValid(wx, wy) &&
                    world.getBoard().getCellType(wx, wy) == CellType::BREAKABLE_WALL) {
                    wallTargetCmd = curr.firstCmd;
                    break;
                }
            }
        }

        if (curr.dist >= 12) continue;

        for (const auto& d : orderedDirs) {
            int nx = curr.x + d.dx;
            int ny = curr.y + d.dy;
            if (world.isPositionValid(nx, ny) && !visited[nx][ny] &&
                canStepOn(nx, ny, world) && !isTileDangerous(nx, ny, world)) {
                visited[nx][ny] = true;
                reachableSafeCount++;
                q.push({nx, ny, curr.firstCmd, curr.dist + 1});
            }
        }
    }

    // If already positioned next to a breakable wall with a valid escape route (waiting on warmup,
    // bomb cooldown, or a nearby bomb to clear) and no power-up is reachable, hold position instead of pacing.
    bool currentlyNearWall = false;
    for (const auto& d : baseDirs) {
        int wx = cx + d.dx;
        int wy = cy + d.dy;
        if (world.isPositionValid(wx, wy) &&
            world.getBoard().getCellType(wx, wy) == CellType::BREAKABLE_WALL) {
            currentlyNearWall = true;
            break;
        }
    }
    if (currentlyNearWall && findEscapeDirection(world, cx, cy, getBombRadius()) != Command::NONE) {
        return Command::NONE;
    }

    Command primaryTargetCmd = (wallTargetCmd != Command::NONE) ? wallTargetCmd : enemyTargetCmd;

    // If temporarily confined to a small safe pocket (<= 3 tiles) with no targets inside
    // (e.g. waiting for another player's bomb blast to clear), stand still instead of pacing back and forth.
    if (primaryTargetCmd == Command::NONE && reachableSafeCount <= 3) {
        return Command::NONE;
    }

    if (primaryTargetCmd != Command::NONE) {
        if (std::find(forwardSafeMoves.begin(), forwardSafeMoves.end(), primaryTargetCmd) != forwardSafeMoves.end() ||
            forwardSafeMoves.empty()) {
            return primaryTargetCmd;
        }
    }

    if (forwardSafeMoves.size() == 1) {
        return forwardSafeMoves[0];
    }

    if (!forwardSafeMoves.empty()) {
        if (std::find(forwardSafeMoves.begin(), forwardSafeMoves.end(), m_currentDir) != forwardSafeMoves.end() &&
            (std::rand() % 100 < 85)) {
            return m_currentDir;
        }
        return forwardSafeMoves[std::rand() % forwardSafeMoves.size()];
    }

    return safeMoves[0];
}

void DumbBot::update(World& world) {
    Player::update(world);
    m_ticksAlive++;
    if (m_bombCooldown > 0) m_bombCooldown--;

    if (!isAlive() || isOnCooldown()) return;

    int cx = getX();
    int cy = getY();
    bool amIDangerous = isTileDangerous(cx, cy, world);

    if (amIDangerous) {
        Command escapeDir = findEscapeDirection(world);
        if (escapeDir != Command::NONE) {
            m_currentDir = escapeDir;
            handleCommand(m_currentDir, world);
            return;
        }

        // If trapped by an adjacent bomb and we have Hammer or Ring, kick or cross the bomb to escape
        struct DirInfo { int dx; int dy; Command cmd; };
        const DirInfo dirs[4] = {
            {0, -1, Command::UP},
            {0, 1, Command::DOWN},
            {-1, 0, Command::LEFT},
            {1, 0, Command::RIGHT}
        };
        if (hasHammer() || hasRing()) {
            for (const auto& d : dirs) {
                int nx = cx + d.dx;
                int ny = cy + d.dy;
                if (!world.isPositionValid(nx, ny)) continue;
                for (const auto* e : world.getEntitiesAt(nx, ny)) {
                    if (e->isAlive() && e->getType() == EntityType::BOMB) {
                        bool canKick = hasHammer() && Bomb::canSlideTo(world, nx + d.dx, ny + d.dy);
                        bool canCross = hasRing() && !canKick;
                        if (canKick || canCross) {
                            m_currentDir = d.cmd;
                            handleCommand(m_currentDir, world);
                            return;
                        }
                    }
                }
            }
        }

        // Otherwise step to the neighbor with the longest survival time
        int bestTte = getTileTimeToExplode(cx, cy, world);
        Command bestCmd = Command::NONE;
        int moveCd = getMoveCooldownMax();
        for (const auto& d : dirs) {
            int nx = cx + d.dx;
            int ny = cy + d.dy;
            if (canStepOn(nx, ny, world)) {
                int tte = getTileTimeToExplode(nx, ny, world);
                if (tte > bestTte || (tte == bestTte && d.cmd == m_currentDir && bestTte > moveCd)) {
                    bestTte = tte;
                    bestCmd = d.cmd;
                }
            }
        }
        if (bestCmd != Command::NONE) {
            m_currentDir = bestCmd;
            handleCommand(m_currentDir, world);
        }
        return;
    }

    // Offensive Hammer kick: if an adjacent bomb can be kicked toward an aligned enemy or wall, kick it
    if (hasHammer()) {
        struct KickDir { int dx; int dy; Command cmd; };
        const KickDir kDirs[4] = {
            {0, -1, Command::UP},
            {0, 1, Command::DOWN},
            {-1, 0, Command::LEFT},
            {1, 0, Command::RIGHT}
        };
        for (const auto& d : kDirs) {
            int bx = cx + d.dx;
            int by = cy + d.dy;
            if (!world.isPositionValid(bx, by) || !Bomb::canSlideTo(world, bx + d.dx, by + d.dy)) continue;
            bool hasBombAtB = false;
            for (const auto* e : world.getEntitiesAt(bx, by)) {
                if (e->isAlive() && e->getType() == EntityType::BOMB) {
                    hasBombAtB = true;
                    break;
                }
            }
            if (!hasBombAtB) continue;

            bool targetAhead = false;
            for (int dist = 2; dist <= 6; ++dist) {
                int tx = cx + d.dx * dist;
                int ty = cy + d.dy * dist;
                if (!world.isPositionValid(tx, ty)) break;
                CellType ct = world.getBoard().getCellType(tx, ty);
                if (ct == CellType::SOLID_WALL) break;
                if (ct == CellType::BREAKABLE_WALL) {
                    targetAhead = true;
                    break;
                }
                for (const auto* e : world.getEntitiesAt(tx, ty)) {
                    if (e->isAlive() && e->getType() == EntityType::PLAYER && e != this) {
                        targetAhead = true;
                        break;
                    }
                }
                if (targetAhead) break;
            }
            if (targetAhead) {
                m_currentDir = d.cmd;
                handleCommand(m_currentDir, world);
                return;
            }
        }
    }

    // Evaluate strategic bomb placement only when safe, warmed up, off bomb cooldown, and having no active bomb out
    if (m_ticksAlive >= SPAWN_BOMB_WARMUP_TICKS &&
        m_bombCooldown == 0 &&
        canPlaceBomb(world) &&
        world.countBombsForOwner(getId()) == 0) {

        // Avoid dropping a second bomb right next to another ticking bomb (prevents multi-bot corridor corking)
        bool bombNearby = false;
        for (const auto& entity : world.getEntities()) {
            if (entity->isAlive() && entity->getType() == EntityType::BOMB) {
                int dist = std::abs(entity->getX() - cx) + std::abs(entity->getY() - cy);
                if (dist <= 4) {
                    bombNearby = true;
                    break;
                }
            }
        }

        if (!bombNearby) {
            auto isBreakableWall = [&](int nx, int ny) {
                if (!world.isPositionValid(nx, ny)) return false;
                return world.getBoard().getCellType(nx, ny) == CellType::BREAKABLE_WALL;
            };

            bool nearWall = isBreakableWall(cx + 1, cy) || isBreakableWall(cx - 1, cy) ||
                            isBreakableWall(cx, cy + 1) || isBreakableWall(cx, cy - 1);

            bool nearEnemy = false;
            for (const auto& entity : world.getEntities()) {
                if (entity->isAlive() && entity->getType() == EntityType::PLAYER && entity.get() != this) {
                    int dist = std::abs(entity->getX() - cx) + std::abs(entity->getY() - cy);
                    if (dist <= 2) {
                        nearEnemy = true;
                        break;
                    }
                }
            }

            if (nearWall || nearEnemy) {
                Command escapeDir = findEscapeDirection(world, cx, cy, getBombRadius());
                if (escapeDir != Command::NONE) {
                    handleCommand(Command::PLACE_BOMB, world);
                    m_bombCooldown = BOMB_PLACEMENT_COOLDOWN;
                    m_currentDir = escapeDir;
                    handleCommand(m_currentDir, world);
                    return;
                }
            }
        }
    }

    Command moveDir = findExplorationDirection(world);
    if (moveDir != Command::NONE) {
        m_currentDir = moveDir;
        handleCommand(m_currentDir, world);
    } else {
        m_currentDir = Command::NONE;
    }
}

void Flame::update(World& /*world*/) {
    if (!isAlive()) return;

    m_activeForCollision = true;

    if (m_timer > 0) {
        m_timer--;
        if (m_timer == 0) {
            destroy();
        }
    }

    // Flames pass through other flames without being destroyed
}

void Flame::onIntersect(Entity& other, World& world) {
    if (!isAlive() || !other.isAlive() || !m_activeForCollision) return;

    if (other.getType() != EntityType::FLAME) {
        bool isBreakable = (other.getType() == EntityType::BREAKABLE_WALL);
        bool isPlayer = (other.getType() == EntityType::PLAYER);
        int victimId = isPlayer ? static_cast<Player&>(other).getId() : -1;
        int livesBefore = isPlayer ? static_cast<Player&>(other).getLives() : 0;

        if (other.getType() == EntityType::BOMB && m_ownerId > 0) {
            static_cast<Bomb&>(other).push(0, 0, m_ownerId);
        }

        other.onExplosionHit(world);

        if (isBreakable && !other.isAlive() && m_ownerId > 0) {
            world.recordWallDestroyed(m_ownerId);
        }
        if (isPlayer && m_ownerId > 0 && m_ownerId != victimId) {
            int livesAfter = other.isAlive() ? static_cast<Player&>(other).getLives() : 0;
            if (livesAfter < livesBefore) {
                world.recordEnemyHit(m_ownerId);
            }
            if (!other.isAlive()) {
                world.recordEnemyKilled(m_ownerId);
            }
        }
    }
}

EntityType Player::getType() const { return EntityType::PLAYER; }

char Player::getDisplayChar() const {
    if (m_invincibilityTimer > 0 && (m_invincibilityTimer % 4 < 2)) {
        return ' ';
    }
    return 'P';
}

void Player::update(World& /*world*/) {
    if (m_cooldown > 0) m_cooldown--;
    if (m_invincibilityTimer > 0) m_invincibilityTimer--;
}

void Player::onExplosionHit(World& world) {
    onExplosionHit(world, false);
}

void Player::onExplosionHit(World& /*world*/, bool bypassInvincibility) {
    if (!bypassInvincibility && m_invincibilityTimer > 0) return;
    if (m_lives > 0) {
        m_lives--;
    }
    if (m_lives > 0) {
        m_invincibilityTimer = INVINCIBILITY_FRAMES;
    } else {
        destroy();
    }
}

void Player::tryPlaceOrPushBomb(World& world) {
    int cx = getX();
    int cy = getY();

    // Standing on a bomb: do nothing (do not place duplicate bomb on the same tile)
    for (auto* e : world.getEntitiesAt(cx, cy)) {
        if (e->isAlive() && e->getType() == EntityType::BOMB) {
            return;
        }
    }

    if (canPlaceBomb(world)) {
        world.addEntity(std::make_unique<Bomb>(cx, cy, DEFAULT_BOMB_TIMER, m_bombRadius, m_id));
    }
}

void Player::tryMove(int targetX, int targetY, World& world) {
    bool canMove = world.isTileFree(targetX, targetY);

    if (!canMove && world.isPositionValid(targetX, targetY)) {
        auto entities = world.getEntitiesAt(targetX, targetY);

        if (m_hasHammer && (m_facingDx != 0 || m_facingDy != 0)) {
            int destX = targetX + m_facingDx;
            int destY = targetY + m_facingDy;
            if (Bomb::canSlideTo(world, destX, destY)) {
                for (auto* e : entities) {
                    if (!e->isAlive()) continue;
                    if (auto* pushable = dynamic_cast<IPushable*>(e)) {
                        pushable->push(m_facingDx, m_facingDy, m_id);
                        return;
                    }
                }
            }
        }

        if (m_hasRing && world.getBoard().getCellType(targetX, targetY) == CellType::EMPTY) {
            canMove = true;
            for (auto* e : entities) {
                if (e->isAlive() && e->isSolid() && e->getType() != EntityType::BOMB) {
                    canMove = false;
                    break;
                }
            }
        }
    }

    if (canMove) {
        setX(targetX);
        setY(targetY);
        m_cooldown = getMoveCooldownMax();
    }
}

void Player::handleCommand(Command command, World& world) {
    if (!isAlive()) return;

    if (command == Command::PLACE_BOMB) {
        tryPlaceOrPushBomb(world);
        return;
    }

    int newDx = 0;
    int newDy = 0;
    switch (command) {
        case Command::UP:    newDx = 0;  newDy = -1; break;
        case Command::DOWN:  newDx = 0;  newDy = 1;  break;
        case Command::LEFT:  newDx = -1; newDy = 0;  break;
        case Command::RIGHT: newDx = 1;  newDy = 0;  break;
        default: return;
    }

    int cx = getX();
    int cy = getY();
    int targetX = cx + newDx;
    int targetY = cy + newDy;

    if (isOnCooldown()) {
        return;
    }

    m_facingDx = newDx;
    m_facingDy = newDy;

    tryMove(targetX, targetY, world);
}

ClassicLevel::ClassicLevel(int numPlayers, int boardWidth, int boardHeight, int wallDensityPercent)
    : m_numPlayers(numPlayers), m_wallDensityPercent(std::clamp(wallDensityPercent, 10, 85)) {
    if (numPlayers >= 1) m_spawnPoints.push_back({0, 0});
    if (numPlayers >= 2) m_spawnPoints.push_back({boardWidth - 1, boardHeight - 1});
    if (numPlayers >= 3) m_spawnPoints.push_back({0, boardHeight - 1});
    if (numPlayers >= 4) m_spawnPoints.push_back({boardWidth - 1, 0});
}

std::vector<Position> ClassicLevel::getSpawnPoints() const {
    return m_spawnPoints;
}

bool ClassicLevel::isSafeZone(int x, int y) const {
    for (const auto& spawnPoint : m_spawnPoints) {
        int manhattanDistance = std::abs(spawnPoint.x - x) + std::abs(spawnPoint.y - y);
        if (manhattanDistance <= 1) {
            return true;
        }
    }
    return false;
}

void ClassicLevel::generate(World& world) {
    Board& board = world.getBoard();
    int width = board.getWidth();
    int height = board.getHeight();

    for (int x = 0; x < width; x++) {
        for (int y = 0; y < height; y++) {
            if ((x % 2) != 0 && (y % 2) != 0) {
                board.setCellType(x, y, CellType::SOLID_WALL);
            }
        }
    }

    for (int x = 0; x < width; x++) {
        for (int y = 0; y < height; y++) {
            if (board.getCellType(x, y) == CellType::SOLID_WALL) {
                continue;
            }
            if (std::rand() % 100 < m_wallDensityPercent) {
                if (!isSafeZone(x, y)) {
                    board.setCellType(x, y, CellType::BREAKABLE_WALL);
                }
            }
        }
    }
    world.sync();
}

#include <iostream>
#include <ctime>
#include <cstdlib>
#include <thread>

#include <limits>
#include <algorithm>

#include "../include/core/Game.h"

#include "../include/core/ConsoleInput.h"

namespace {
    constexpr int BOARD_WIDTH = 13;
    constexpr int BOARD_HEIGHT = 13;
    constexpr int MAX_TOTAL_PARTICIPANTS = 4;
    constexpr int MAX_HUMAN_PLAYERS = 2;
    constexpr int TARGET_FPS = 30;
    constexpr int FRAME_DURATION_MS = 1000 / TARGET_FPS;
}

struct GameConfig {
    int numPlayers = 1;
    int numBots = 1;
};

int readBoundedInt(const char* prompt, int minVal, int maxVal) {
    int value = minVal;
    std::cout << prompt;
    if (!(std::cin >> value)) {
        std::cin.clear();
        std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
        value = minVal;
    }
    return std::clamp(value, minVal, maxVal);
}

GameConfig showMenu() {
    GameConfig config;
    std::cout << "\033[2J\033[1;1H";
    std::cout << "=================================\n";
    std::cout << "         BOMBERMAN CLONE         \n";
    std::cout << "=================================\n\n";

    config.numPlayers = readBoundedInt("Enter number of human players (0 - 2): ", 0, MAX_HUMAN_PLAYERS);

    int maxBots = MAX_TOTAL_PARTICIPANTS - config.numPlayers;
    int minBots = (config.numPlayers == 0) ? 2 : 0;
    std::string botPrompt = "Enter number of bots (" + std::to_string(minBots) + " - " + std::to_string(maxBots) + "): ";
    config.numBots = readBoundedInt(botPrompt.c_str(), minBots, maxBots);

    return config;
}

int main() {
    srand(static_cast<unsigned int>(time(nullptr)));

    GameConfig config = showMenu();
    int totalParticipants = std::max(1, config.numPlayers + config.numBots);

    Game game(BOARD_WIDTH, BOARD_HEIGHT);
    ConsoleRenderer renderer;
    ConsoleInput consoleInput;

    ClassicLevel level(totalParticipants, BOARD_WIDTH, BOARD_HEIGHT);
    game.loadLevel(level, config.numPlayers, config.numBots);

    bool isRunning = true;

    while (isRunning) {
        auto frameStart = std::chrono::high_resolution_clock::now();

        auto commands = consoleInput.pollCommands();
        for (const auto& pc : commands) {
            if (pc.command == Command::QUIT) {
                isRunning = false;
                break;
            }
        }
        if (!isRunning) {
            break;
        }

        game.update(commands);
        
        // Pass Sudden Death timer and active status to the console renderer
        renderer.render(game.getWorld(), game.getSuddenDeathTimer(), game.isSuddenDeathActive());

        if (game.isGameOver()) {
            break;
        }

        auto frameEnd = std::chrono::high_resolution_clock::now();
        auto frameDuration = std::chrono::duration_cast<std::chrono::milliseconds>(frameEnd - frameStart).count();

        int sleepTime = FRAME_DURATION_MS - static_cast<int>(frameDuration);
        if (sleepTime > 0) {
            std::this_thread::sleep_for(std::chrono::milliseconds(sleepTime));
        }
    }

    int winnerId = game.getWinnerId();
    std::cout << "\n============================================\n";
    if (winnerId == -1) {
        std::cout << "               IT'S A DRAW!                 \n";
    } else {
        const Player* winner = game.getWorld().getPlayer(winnerId);
        if (winner && winner->isBot()) {
            std::cout << "               BOT " << winnerId << " WINS!                 \n";
        } else {
            std::cout << "             PLAYER " << winnerId << " WINS!                \n";
        }
    }
    std::cout << "============================================\n";

    return 0;
}

void CollisionSystem::process(World& world) {
    const auto& entities = world.getEntities();
    
    for (size_t i = 0; i < entities.size(); ++i) {
        if (!entities[i]->isAlive()) continue;
        
        for (size_t j = i + 1; j < entities.size(); ++j) {
            if (!entities[j]->isAlive()) continue;
            
            if (entities[i]->getX() == entities[j]->getX() && 
                entities[i]->getY() == entities[j]->getY()) {
                
                entities[i]->onIntersect(*entities[j], world);
                entities[j]->onIntersect(*entities[i], world);
            }
        }
    }
}

#include "../../include/systems/ExplosionSystem.h"
#include "../../include/core/World.h"
#include "../../include/entities/Flame.h"
#include "../../include/entities/Bomb.h"
#include "../../include/entities/Player.h"
#include "../../include/factories/PowerUpFactory.h"

void ExplosionSystem::process(World& world) {
    bool hasExplodingBombs = true;
    int maxIterations = static_cast<int>(world.getEntities().size()) + 16;

    while (hasExplodingBombs && maxIterations-- > 0) {
        hasExplodingBombs = false;

        const auto& entities = world.getEntities();
        Bomb* bombToExplode = nullptr;

        for (const auto& entity : entities) {
            if (entity->isAlive() && entity->shouldExplode() && entity->getType() == EntityType::BOMB) {
                bombToExplode = static_cast<Bomb*>(entity.get());
                break;
            }
        }

        if (bombToExplode) {
            hasExplodingBombs = true;

            int bx = bombToExplode->getX();
            int by = bombToExplode->getY();
            int radius = bombToExplode->getRadius();
            int creditOwnerId = bombToExplode->getCreditOwnerId();

            bombToExplode->destroy();

            explodeTile(world, bx, by, creditOwnerId);

            explodeInDirection(world, bx, by, 1, 0, radius, creditOwnerId);
            explodeInDirection(world, bx, by, -1, 0, radius, creditOwnerId);
            explodeInDirection(world, bx, by, 0, 1, radius, creditOwnerId);
            explodeInDirection(world, bx, by, 0, -1, radius, creditOwnerId);

            world.sync();
        }
    }
}

bool ExplosionSystem::explodeTile(World& world, int targetX, int targetY, int ownerId) {
    if (!world.isPositionValid(targetX, targetY)) {
        return false;
    }

    CellType cellType = world.getBoard().getCellType(targetX, targetY);
    if (cellType == CellType::SOLID_WALL) {
        return false;
    }

    bool explosionStopped = false;
    if (cellType == CellType::BREAKABLE_WALL) {
        world.getBoard().setCellType(targetX, targetY, CellType::EMPTY);
        if (ownerId > 0) {
            world.recordWallDestroyed(ownerId);
        }
        PowerUpFactory::trySpawn(world, targetX, targetY);
        explosionStopped = true;
    }

    auto entitiesHere = world.getEntitiesAt(targetX, targetY);

    for (auto* entity : entitiesHere) {
        if (entity->isAlive()) {
            EntityType type = entity->getType();
            bool isPlayer = (type == EntityType::PLAYER);
            int victimId = isPlayer ? static_cast<Player*>(entity)->getId() : -1;
            int livesBefore = isPlayer ? static_cast<Player*>(entity)->getLives() : 0;
            bool blocksBlast = entity->stopsExplosion();

            if (type == EntityType::BOMB && ownerId > 0) {
                // Propagate credit owner in chain reactions
                static_cast<Bomb*>(entity)->push(0, 0, ownerId);
            }

            entity->onExplosionHit(world);

            if (isPlayer && ownerId > 0 && ownerId != victimId) {
                int livesAfter = entity->isAlive() ? static_cast<Player*>(entity)->getLives() : 0;
                if (livesAfter < livesBefore) {
                    world.recordEnemyHit(ownerId);
                }
                if (!entity->isAlive()) {
                    world.recordEnemyKilled(ownerId);
                }
            }

            if (blocksBlast) {
                explosionStopped = true;
            }
        }
    }

    world.addEntity(std::make_unique<Flame>(targetX, targetY, 30, ownerId));

    return !explosionStopped;
}

void ExplosionSystem::explodeInDirection(World& world, int startX, int startY, int dx, int dy, int radius, int ownerId) {
    for (int i = 1; i <= radius; i++) {
        if (!explodeTile(world, startX + i * dx, startY + i * dy, ownerId)) {
            break;
        }
    }
}

#include "../../include/systems/SuddenDeathSystem.h"
#include "../../include/core/World.h"
#include "../../include/core/Board.h"
#include "../../include/entities/Entity.h"
#include "../../include/entities/Player.h"

SuddenDeathSystem::SuddenDeathSystem(int startDelayTicks, int boardWidth, int boardHeight)
    : m_ticksUntilStart(startDelayTicks),
      m_dropCooldown(15), // 15 frames at 30 FPS = 2 tiles per second
      m_currentCooldown(0),
      m_minX(0), m_maxX(boardWidth - 1),
      m_minY(0), m_maxY(boardHeight - 1),
      m_x(0), m_y(0),
      m_dx(1), m_dy(0),
      m_isFinished(false) {}

int SuddenDeathSystem::getTicksRemaining() const { 
    return m_ticksUntilStart; 
}

void SuddenDeathSystem::setTicksRemaining(int ticks) {
    m_ticksUntilStart = ticks;
}

bool SuddenDeathSystem::isActive() const { 
    return m_ticksUntilStart <= 0 && !m_isFinished; 
}

std::vector<SuddenDeathDangerTile> SuddenDeathSystem::getImminentDangerTiles(int lookaheadTicks) const {
    std::vector<SuddenDeathDangerTile> result;
    if (m_isFinished || m_ticksUntilStart > lookaheadTicks) {
        return result;
    }

    int ticksUntilNext = (m_ticksUntilStart > 0 ? m_ticksUntilStart : 0) + m_currentCooldown;
    int minX = m_minX, maxX = m_maxX;
    int minY = m_minY, maxY = m_maxY;
    int x = m_x, y = m_y;
    int dx = m_dx, dy = m_dy;
    bool finished = m_isFinished;

    while (!finished && ticksUntilNext <= lookaheadTicks) {
        result.push_back({x, y, ticksUntilNext});

        if (dx == 1 && x == maxX) {
            dx = 0; dy = 1; minY++;
        } else if (dx == 0 && dy == 1 && y == maxY) {
            dx = -1; dy = 0; maxX--;
        } else if (dx == -1 && x == minX) {
            dx = 0; dy = -1; maxY--;
        } else if (dx == 0 && dy == -1 && y == minY) {
            dx = 1; dy = 0; minX++;
        }

        x += dx;
        y += dy;

        if (minX > maxX || minY > maxY || (maxX - minX <= 4 && maxY - minY <= 4)) {
            finished = true;
        } else {
            ticksUntilNext += m_dropCooldown + 1;
        }
    }
    return result;
}

void SuddenDeathSystem::process(World& world) {
    if (m_isFinished) return;

    if (m_ticksUntilStart > 0) {
        m_ticksUntilStart--;
        return;
    }

    if (m_currentCooldown > 0) {
        m_currentCooldown--;
        return;
    }

    world.getBoard().setCellType(m_x, m_y, CellType::SOLID_WALL);

    auto crushEntities = [&](const std::vector<std::unique_ptr<Entity>>& list) {
        for (const auto& e : list) {
            if (e->isAlive() && e->getX() == m_x && e->getY() == m_y) {
                if (e->getType() == EntityType::PLAYER) {
                    auto* p = static_cast<Player*>(e.get());
                    int maxHits = p->getLives() + 1;
                    while (p->isAlive() && maxHits-- > 0) {
                        p->onExplosionHit(world, true);
                    }
                    if (p->isAlive()) {
                        p->destroy();
                    }
                } else {
                    e->destroy();
                }
            }
        }
    };
    crushEntities(world.getEntities());
    crushEntities(world.getSpawnQueue());

    if (m_dx == 1 && m_x == m_maxX) {
        m_dx = 0; m_dy = 1; m_minY++;
    } else if (m_dx == 0 && m_dy == 1 && m_y == m_maxY) {
        m_dx = -1; m_dy = 0; m_maxX--;
    } else if (m_dx == -1 && m_x == m_minX) {
        m_dx = 0; m_dy = -1; m_maxY--;
    } else if (m_dx == 0 && m_dy == -1 && m_y == m_minY) {
        m_dx = 1; m_dy = 0; m_minX++;
    }

    m_x += m_dx;
    m_y += m_dy;

    if (m_minX > m_maxX || m_minY > m_maxY || (m_maxX - m_minX <= 4 && m_maxY - m_minY <= 4)) {
        m_isFinished = true;
    } else {
        m_currentCooldown = m_dropCooldown;
    }
}
